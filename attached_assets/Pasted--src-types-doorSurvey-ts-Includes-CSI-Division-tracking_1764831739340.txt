// src/types/doorSurvey.ts

// Includes CSI Division tracking (DIV08, DIV28)
export type BuildingCodeStandard =
  | "ADA"
  | "IBC"
  | "NFPA"
  | "ANSI_SDI"
  | "LOCAL"
  | "DIV08"
  | "DIV28";

export interface DoorSurveyProject {
  id?: string;
  name: string;
  location: string;
  customerId?: string;
  buildingType: string;
  quoteType: "none" | "budget" | "proposal" | "change_order";
  standards: BuildingCodeStandard[]; // now can include DIV08 + DIV28
}

export interface Measurements {
  roughWidthMm: number;
  roughHeightMm: number;
  clearWidthMm: number;
  headHeightMm: number;
  jambDepthMm: number;
  doorThicknessMm: number;
  sillHeightMm?: number;
}

export type OpeningType =
  | "single"
  | "pair"
  | "sliding"
  | "folding"
  | "revolving"
  | "other";

export type Handing = "RH" | "LH" | "RHR" | "LHR";

export interface DoorLeaf {
  leafType: "flush" | "full_glass" | "half_glass" | "narrow_lite";
  coreType: "solid" | "hollow" | "mineral" | "stc";
  fireRatingMinutes: number;
  hasVisionLite: boolean;
  visionLiteDescription?: string;
  manufacturer?: string;
  model?: string;
}

export interface Frame {
  material: "hollow_metal" | "aluminum" | "wood";
  profile: string;
  anchorType: "existing" | "new" | "steel";
  conditionScore: 1 | 2 | 3 | 4 | 5;
  notes?: string;
}

export interface HardwarePiece {
  type:
    | "hinge"
    | "lockset"
    | "exit_device"
    | "closer"
    | "strike"
    | "electric_strike"
    | "mag_lock"
    | "operator"
    | "accessory";
  description: string;
  manufacturer?: string;
  model?: string;
  qty?: number;
  notes?: string;
}

// NEW – Division 28 devices
export interface ElectronicSecurityDevice {
  type:
    | "card_reader"
    | "keypad"
    | "door_contact"
    | "rex_device"
    | "request_to_exit_bar"
    | "camera"
    | "intercom"
    | "power_supply"
    | "fire_alarm_interface"
    | "other";
  description: string;
  manufacturer?: string;
  model?: string;
  qty?: number;
  monitored?: boolean;
  notes?: string;
}

export interface HardwareSet {
  hinges: HardwarePiece[];
  lockFunction: string; // e.g. 45 – storeroom
  lockset: string;
  exitDevice?: string;
  closer?: string;
  accessories: HardwarePiece[];
  electrifiedHardware: HardwarePiece[];
  // NEW – Division 28
  electronicSecurity: ElectronicSecurityDevice[];
}

export interface ComplianceIssue {
  // Can refer to DIV08 or DIV28 as well as codes
  standard: BuildingCodeStandard;
  codeRef: string; // e.g. "ADA 404.2.3" or "DIV28 Access Control"
  severity: "info" | "warning" | "critical";
  message: string;
}

export interface ComplianceResult {
  standardsChecked: BuildingCodeStandard[];
  passes: boolean;
  issues: ComplianceIssue[];
  notes?: string;
}

export interface EvidenceFile {
  id?: string;
  type: "photo" | "video" | "document";
  url: string;
  label?: string;
}

export interface Evidence {
  photos: EvidenceFile[];
  videos: EvidenceFile[];
  documents: EvidenceFile[];
}

export interface EngineeringConflict {
  type:
    | "hardware"
    | "fire_rating"
    | "egress"
    | "structure"
    | "electrical"
    | "security";
  message: string;
}

export interface EngineeringReview {
  status: "pending" | "approved" | "changes_required";
  notes?: string;
  conflicts: EngineeringConflict[];
}

export interface QuoteSummary {
  materialCost: number;
  laborCost: number;
  shippingCost: number;
  marginPercent: number;
  totalPrice: number;
  optionLabel: string; // e.g. "Base", "Option A"
}

export interface DoorOpeningInput {
  id?: string;
  doorTag: string;
  floor: string;
  spaceName: string;
  openingType: OpeningType;
  handing: Handing;
  measurements: Measurements;
  doorLeaf: DoorLeaf;
  frame: Frame;
  hardware: HardwareSet;
  compliance: ComplianceResult;
  evidence: Evidence;
  engineeringReview: EngineeringReview;
  quoteSummary: QuoteSummary;
}

export interface DoorSurveyPayload {
  project: DoorSurveyProject;
  opening: DoorOpeningInput;
}
