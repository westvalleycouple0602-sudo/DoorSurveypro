/**
 * Core rule model for door, frame, and hardware checks.
 * You can extend this with more jurisdictions, versions, etc.
 */

export type CodeSource =
  | "ADA_2010"
  | "IBC_2018"
  | "IBC_2021"
  | "NFPA_80"
  | "NFPA_101"
  | "ICC_A117_1"
  | "BHMA_A156";

export type RuleCategory =
  | "operable_parts_egress"
  | "accessibility_reach_operability"
  | "fire_door_assembly"
  | "vision_communication"
  | "automatic_operators"
  | "thresholds_clearances"
  | "hinges_pivots_misc";

export interface CodeReference {
  source: CodeSource;
  section: string; // e.g. "404.2.7", "1010.2"
  notes?: string;
}

export interface HeightRequirement {
  /** Minimum height in inches above finished floor (AFF). */
  min_in?: number;
  /** Maximum height in inches above finished floor (AFF). */
  max_in?: number;
}

export interface DimensionalRequirement {
  /** Maximum threshold height in inches. */
  threshold_max_height_in?: number;
  /** Maximum vertical edge clearance between door and frame (inches). */
  max_vertical_clearance_in?: number;
  /** Maximum bottom clearance (inches) for fire doors, if applicable. */
  max_bottom_clearance_in?: number;
}

export interface RuleRequirement {
  /** AFF height requirement for the relevant operable part, bar, plate, viewer, etc. */
  height_aff?: HeightRequirement;
  /** Other dimensional requirements related to this rule. */
  dimensions?: DimensionalRequirement;
  /** Flags/expectations this rule asserts (non-numeric). */
  flags?: {
    /** Hardware must unlatch with a single motion. */
    single_motion_to_egress?: boolean;
    /** Operation must not require tight grasping, pinching, or twisting of the wrist. */
    no_tight_grasp_pinch_twist?: boolean;
    /** Hardware must be operable with one hand. */
    one_hand_operation?: boolean;
    /** Assembly must be listed and labeled for fire use. */
    fire_listed_assembly_required?: boolean;
    /** Door must be self-closing or automatic-closing if fire rated. */
    self_closing_required_for_fire_door?: boolean;
  };
}

/**
 * A single rule that your engine can evaluate against a surveyed opening.
 */
export interface DoorHardwareRule {
  /** Unique stable id; you can map violations back to this. */
  id: string;
  /** Human-readable rule name. */
  name: string;
  /** Short description of what the rule is about. */
  description: string;
  /** High-level category for grouping in UI/analytics. */
  category: RuleCategory;
  /** Typical jurisdictions this rule applies to (you can add more granular configs). */
  jurisdictions: string[]; // e.g. ["US"], ["US-CA"]
  /** Formal code references that justify the rule. */
  code_references: CodeReference[];
  /** The actual numeric/dimensional and functional requirements. */
  requirements: RuleRequirement;
  /**
   * Machine-readable condition string(s) describing when the rule applies.
   * You can parse this or just show it in the UI and implement logic separately.
   */
  conditions: string[];
  /** Additional implementation notes or nuance for engineers/reviewers. */
  notes?: string;
}

/**
 * Utility helpers you can use in your services.
 */

export function getRulesByCategory(category: RuleCategory): DoorHardwareRule[] {
  return doorHardwareRules.filter((r) => r.category === category);
}

export function getRulesForJurisdiction(juris: string): DoorHardwareRule[] {
  return doorHardwareRules.filter(
    (r) => r.jurisdictions.includes(juris) || r.jurisdictions.includes("US")
  );
}

export function findRuleById(id: string): DoorHardwareRule | undefined {
  return doorHardwareRules.find((r) => r.id === id);
}
